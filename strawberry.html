import pygame import random import sys # 初期化 pygame.init() # 画面サイズ WIDTH, HEIGHT = 800, 600 screen = pygame.display.set_mode((WIDTH, HEIGHT)) pygame.display.set_caption("ロフィアくんのキャッチゲーム") # 色 WHITE = (255, 255, 255) BLACK = (0, 0, 0) GRAY = (200, 200, 200) # フォント font = pygame.font.Font(None, 36) # 画像読み込み basket_img = pygame.image.load("basket.png") strawberry_img = pygame.image.load("strawberry.png") rofiakun_img = pygame.image.load("rofiakun (1).png") # キャラクターのアスペクト比固定（0.517:1） CHARACTER_WIDTH = 120 CHARACTER_HEIGHT = int(CHARACTER_WIDTH / 0.517) rofiakun_img = pygame.transform.scale(rofiakun_img, (CHARACTER_WIDTH, CHARACTER_HEIGHT)) # キャラの設定 character_x = WIDTH // 2 - CHARACTER_WIDTH // 2 character_y = HEIGHT - CHARACTER_HEIGHT - 50 character_speed = 7 # かご設定 basket_width, basket_height = 100, 60 basket_img = pygame.transform.scale(basket_img, (basket_width, basket_height)) # いちご設定 strawberry_width, strawberry_height = 50, 50 strawberry_img = pygame.transform.scale(strawberry_img, (strawberry_width, strawberry_height)) fall_speed = 5 strawberries = [] score = 0 # ボタンエリア button_height = 80 button_left = pygame.Rect(50, HEIGHT - button_height, 150, button_height) button_right = pygame.Rect(WIDTH - 200, HEIGHT - button_height, 150, button_height) # いちご生成 def create_strawberry(): x = random.randint(0, WIDTH - strawberry_width) y = -strawberry_height strawberries.append(pygame.Rect(x, y, strawberry_width, strawberry_height)) # ゲームループ clock = pygame.time.Clock() spawn_event = pygame.USEREVENT + 1 pygame.time.set_timer(spawn_event, 1500) running = True while running: screen.fill(WHITE) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == spawn_event: create_strawberry() elif event.type == pygame.MOUSEBUTTONDOWN: if button_left.collidepoint(event.pos): character_x -= character_speed * 5 # 長押し代わりに強め移動 elif button_right.collidepoint(event.pos): character_x += character_speed * 5 # キーボード操作 keys = pygame.key.get_pressed() if keys[pygame.K_LEFT]: character_x -= character_speed if keys[pygame.K_RIGHT]: character_x += character_speed # 画面内制限 character_x = max(0, min(WIDTH - CHARACTER_WIDTH, character_x)) # キャラとカゴ描画 screen.blit(rofiakun_img, (character_x, character_y)) screen.blit(basket_img, (character_x + CHARACTER_WIDTH // 2 - basket_width // 2, character_y + CHARACTER_HEIGHT - 20)) basket_rect = pygame.Rect(character_x + CHARACTER_WIDTH // 2 - basket_width // 2, character_y + CHARACTER_HEIGHT - 20, basket_width, basket_height) # いちご更新 for strawberry in strawberries[:]: strawberry.y += fall_speed screen.blit(strawberry_img, strawberry.topleft) if strawberry.colliderect(basket_rect): strawberries.remove(strawberry) score += 1 elif strawberry.y > HEIGHT: strawberries.remove(strawberry) # スコア表示 score_text = font.render(f"Score: {score}", True, BLACK) screen.blit(score_text, (10, 10)) # ボタン描画 pygame.draw.rect(screen, GRAY, button_left) pygame.draw.rect(screen, GRAY, button_right) left_text = font.render("←", True, BLACK) right_text = font.render("→", True, BLACK) screen.blit(left_text, (button_left.centerx - 10, button_left.centery - 15)) screen.blit(right_text, (button_right.centerx - 10, button_right.centery - 15)) pygame.display.flip() clock.tick(60) pygame.quit() sys.exit()
